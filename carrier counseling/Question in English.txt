let interestScores = {
    "Math": 0,
    "Biology": 0,
    "Commerce": 0,
    "Arts": 0
};

let askedQuestions = new Set();
let currentInterest = "";

const questionContainer = document.getElementById("question");
const resultContainer = document.getElementById("result");
const yesButton = document.getElementById("yesBtn");
const noButton = document.getElementById("noBtn");

function getRandomQuestion(interest) {
    let availableQuestions = questions[interest].filter(q => !askedQuestions.has(q));
    if (availableQuestions.length === 0) {
        return null;
    }
    let question = availableQuestions[Math.floor(Math.random() * availableQuestions.length)];
    return question;
}

function displayQuestion() {
    currentInterest = Object.keys(questions)[Math.floor(Math.random() * Object.keys(questions).length)];
    let question = getRandomQuestion(currentInterest);
    if (question) {
        questionContainer.innerText = question;
    } else {
        showResult();
    }
}

function processAnswer(answer) {
    if (answer === "yes") {
        interestScores[currentInterest]++;
    }
    askedQuestions.add(questionContainer.innerText);
    displayQuestion();
}


function showResult() {
    questionContainer.style.display = "none";
    yesButton.style.display = "none";
    noButton.style.display = "none";
    resultContainer.innerHTML = "<h2>Based on your interests, here are some suggested subjects:</h2>";
    let maxInterest = Object.keys(interestScores).reduce((a, b) => interestScores[a] > interestScores[b] ? a : b);
    if (interestScores[maxInterest] > 0) {
        resultContainer.innerHTML += `<p class="txt2">- Your main interest is ${maxInterest}!</p>`;
        resultContainer.innerHTML += `<p class="txt2">Do you want to know about career options in ${maxInterest}? (yes/no)</p>`;
        let answerBtns = document.createElement("div");
        answerBtns.innerHTML = `<button class="button" id="careerYesBtn">Yes</button><button class="button" id="careerNoBtn">No</button>`;
        resultContainer.appendChild(answerBtns);
        document.getElementById("careerYesBtn").addEventListener("click", () => {
            showCareerOptions(maxInterest);
        });
        document.getElementById("careerNoBtn").addEventListener("click", () => {
            resultContainer.innerHTML += "<p>No specific interest found, try exploring more!</p>";
        });
    }
}

function showCareerOptions(interest) {
    let careerOptions = {
        "Math": ["Computer Science", "Cryptograpy", "Mathematical Modelling", "Operations Research", "Financial Analysis", "Risk Management", "Mathematical Education", "Mathematical Research", "Software Development", "Numerical Analysis"],
        "Biology": ["Microbiology", "Genetics", "Ecology", "Marine Biology", "Zoology", "Botany", "Pharmacology", "Veterinary Science", "Forensic Science", "Bioinformatics"],
        "Commerce": ["Human Resource Management", "Supply Chain Management", "International Business", "Retail Management", "Business Analytics", "Risk Management", "Entrepreneurship", "Corporate Law", "Real Estate Management", "Operations Management"],
        "Arts": ["Fashion Designing", "Interior Designing", "Photography", "Film Making", "Animation", "Creative Writing", "Advertising", "Multimedia Arts", "Illustration", "Theater and Drama"]
    };
    resultContainer.innerHTML = `<h2 class="txt1">Here are some career options in ${interest}:</h2>`;
    let optionsList = document.createElement("ul");
    careerOptions[interest].forEach(option => {
        let listItem = document.createElement("li");
        listItem.textContent = option;
        optionsList.appendChild(listItem);
    });
    resultContainer.appendChild(optionsList);
}

yesButton.addEventListener("click", () => processAnswer("yes"));
noButton.addEventListener("click", () => processAnswer("no"));
