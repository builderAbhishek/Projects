// Function to retrieve student details based on provided class, section, and roll number
function getStudentDetails($conn, $class, $section, $roll_number) {
    $sql = "SELECT * FROM students WHERE class='$class' AND section='$section' AND roll_number='$roll_number'";
    $result = $conn->query($sql);

    if ($result->num_rows > 0) {
        return $result->fetch_assoc();
    } else {
        return false;
    }
}
